cmake_minimum_required(VERSION 3.25)
project(GantriaPhysicsFoundation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(GANTRIA_JOLT_ROOT "" CACHE PATH "Isolated source-bound Jolt checkout")
set(GANTRIA_JOLT_COMMIT "" CACHE STRING "Required exact source identity")
option(GANTRIA_NUMERICS "Use maintained per-body numerical correction" ON)
option(GANTRIA_OBSERVER "Use separately patched internal observer" OFF)
option(GANTRIA_PROFILE "Build separate external-stage profiling binary" OFF)
option(GANTRIA_CONVERGENCE "Compile isolated retained-lambda convergence API" OFF)
if(GANTRIA_CONVERGENCE AND NOT GANTRIA_NUMERICS)
    message(FATAL_ERROR "[Physics:Foundation] Convergence requires maintained numerics")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${GANTRIA_JOLT_ROOT}" rev-parse HEAD
    OUTPUT_VARIABLE Actual OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT GANTRIA_JOLT_COMMIT OR NOT Actual STREQUAL GANTRIA_JOLT_COMMIT)
    message(FATAL_ERROR "[Physics:Foundation] Exact Jolt source identity required")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${GANTRIA_JOLT_ROOT}" status --porcelain
    OUTPUT_VARIABLE Dirty OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(GANTRIA_OBSERVER)
    if(NOT Dirty STREQUAL "M Jolt/Physics/SoftBody/SoftBodyMotionProperties.cpp" AND
       NOT Dirty STREQUAL " M Jolt/Physics/SoftBody/SoftBodyMotionProperties.cpp")
        message(FATAL_ERROR "[Physics:Foundation] Observer requires only its recorded source patch")
    endif()
elseif(Dirty)
    message(FATAL_ERROR "[Physics:Foundation] Pure library checkout must be clean")
endif()
foreach(Feature USE_SSE4_1 USE_SSE4_2 USE_AVX USE_AVX2 USE_AVX512 USE_LZCNT USE_TZCNT USE_F16C USE_FMADD
        TARGET_UNIT_TESTS TARGET_HELLO_WORLD TARGET_PERFORMANCE_TEST TARGET_SAMPLES TARGET_VIEWER
        JPH_USE_DX12 JPH_USE_VK JPH_USE_MTL JPH_USE_CPU_COMPUTE INTERPROCEDURAL_OPTIMIZATION
        GENERATE_DEBUG_SYMBOLS ENABLE_ALL_WARNINGS FLOATING_POINT_EXCEPTIONS_ENABLED
        DEBUG_RENDERER_IN_DEBUG_AND_RELEASE JPH_BUILD_SHARED_LIBS OVERRIDE_CXX_FLAGS ENABLE_INSTALL)
    set(${Feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(USE_STATIC_MSVC_RUNTIME_LIBRARY OFF CACHE BOOL "" FORCE)
set(CROSS_PLATFORM_DETERMINISTIC ON CACHE BOOL "" FORCE)
set(PROFILER_IN_DEBUG_AND_RELEASE ${GANTRIA_PROFILE} CACHE BOOL "" FORCE)
set(JPH_USE_EXTERNAL_PROFILE ${GANTRIA_PROFILE} CACHE BOOL "" FORCE)
add_subdirectory("${GANTRIA_JOLT_ROOT}/Build" JoltBuild EXCLUDE_FROM_ALL)
get_filename_component(Evaluation "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
get_filename_component(Engine "${Evaluation}/../.." ABSOLUTE)
function(AddFoundationTarget Name Source)
    add_executable(${Name} "${Evaluation}/${Source}" "${Evaluation}/ExperimentSupport.cpp")
    target_include_directories(${Name} PRIVATE "${Engine}/vendor/json/include" "${Engine}/vendor/box3d/include")
    target_link_libraries(${Name} PRIVATE Jolt)
    target_compile_definitions(${Name} PRIVATE PHYSICS_UNIFIED_NATIVE_ONLY=1)
    if(GANTRIA_NUMERICS)
        target_compile_definitions(${Name} PRIVATE PHYSICS_GANTRIA_JOLT=1)
    endif()
    if(GANTRIA_CONVERGENCE)
        target_compile_definitions(${Name} PRIVATE PHYSICS_GANTRIA_CONVERGENCE=1)
    endif()
    if(GANTRIA_OBSERVER)
        target_compile_definitions(${Name} PRIVATE PHYSICS_ENERGY_OBSERVER=1)
    endif()
    if(WIN32)
        target_link_libraries(${Name} PRIVATE psapi)
    endif()
endfunction()
if(GANTRIA_PROFILE)
    AddFoundationTarget(PhysicsJoltSelectiveCouplingBenchmark Optimization/SelectiveCouplingBenchmark.cpp)
    target_sources(PhysicsJoltSelectiveCouplingBenchmark PRIVATE "${Evaluation}/Optimization/ExternalProfile.cpp")
else()
    AddFoundationTarget(PhysicsJoltScheduledCouplingDiagnostic ScheduledCouplingDiagnostic.cpp)
    AddFoundationTarget(PhysicsJoltUnifiedCouplingQualification UnifiedCouplingEvaluation.cpp)
    AddFoundationTarget(PhysicsJoltSelectiveCouplingBenchmark Optimization/SelectiveCouplingBenchmark.cpp)
    target_sources(PhysicsJoltSelectiveCouplingBenchmark PRIVATE "${Evaluation}/Optimization/ExternalProfile.cpp")
endif()
